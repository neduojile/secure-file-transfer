    setAdminMode((current) => !current);

      email: "",
      password: "",

    e.preventDefault();

    if (loading) return;

        : "Signing you in...",

      }
    );

      const result = await signIn("credentials", {
        email: form.email,
        password: form.password,
        redirect: false,

            icon: <XCircle className="h-5 w-5" />,

        return;

      const session = await getSession();

          toast.dismiss(loadingToast);

            description:

          icon: <CheckCircle2 className="h-5 w-5" />,

        router.refresh();

          description:
            "Please use Administrator Login to access the security center.",
          icon: <ShieldCheck className="h-5 w-5" />,
        });

        setLoading(false);
        return;
      }

        case "USER":
        case "LECTURER":
        case "HOD":
        case "EXAM_OFFICER":

          toast.success("Login Successful", {
            description: "Welcome back to SATIM.",
            icon: <CheckCircle2 className="h-5 w-5" />,

          router.push("/dashboard");
          router.refresh();
          return;

          toast.success("Login Successful", {
            description:
              "Welcome to the MITM simulation environment.",
            icon: <CheckCircle2 className="h-5 w-5" />,
          });

          router.push("/attacker/dashboard");
          router.refresh();
          return;

            description:
              "Unable to determine your account permissions.",
            icon: <XCircle className="h-5 w-5" />,

function formatDate(date: Date) {
  return new Intl.DateTimeFormat("en", {
    day: "2-digit",
    month: "short",
    hour: "2-digit",
    minute: "2-digit",
  }).format(date);
}

function formatSize(bytes: number) {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
}

function stateFor(transfer: {
  status: string;
  integrity: string;
  attackAction: string;
}) {
  if (
    transfer.integrity === "COMPROMISED" ||
    transfer.attackAction === "MODIFIED"
  ) {
    return {
      label: "Compromised",
      tone: "red",
      icon: ShieldAlert,
    };
  }

  if (transfer.integrity === "VERIFIED") {
    return {
      label: "Verified",
      tone: "blue",
      icon: CheckCircle2,
    };
  }

  if (
    transfer.status === "IN_TRANSIT" ||
    transfer.status === "VERIFYING"
  ) {
    return {
      label: "In transit",
      tone: "amber",
      icon: Activity,
    };
  }

  return {
    label: transfer.status.replaceAll("_", " "),
    tone: "slate",
    icon: Activity,
  };
}

interface TransitFile {
  id: string;
  fileName: string;
  fileType: string;
  fileSize: number;
  encrypted: boolean;
  status: string;
  sender: string;
  receiver: string;
  createdAt: string;
}

export default function AttackerDashboardPage() {
  const [files, setFiles] = useState<TransitFile[]>([]);
  const [loading, setLoading] = useState(true);
  const [processingId, setProcessingId] = useState("");
  const [search, setSearch] = useState("");

  async function loadFiles() {
    try {
      setLoading(true);

      const res = await fetch("/api/attacker");

      const data = await res.json();

      if (!res.ok) {
        throw new Error(
          data.error || "Unable to load files."
        );
      }

      setFiles(data);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    loadFiles();
  }, []);

  async function forwardFile(id: string) {
    try {
      setProcessingId(id);

      const res = await fetch("/api/attacker/forward", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ id }),
      });

      const data = await res.json();

      if (!res.ok) {
        throw new Error(data.error);
      }

      await loadFiles();
    } catch (err: any) {
      alert(err.message);
    } finally {
      setProcessingId("");
    }
  }

  async function replaceFile(id: string) {
    const confirmed = window.confirm(
      "Replace this file with a tampered copy?"
    );

    if (!confirmed) return;

    try {
      setProcessingId(id);

      const res = await fetch("/api/attacker/replace", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ id }),
      });

      const data = await res.json();

      if (!res.ok) {
        throw new Error(data.error);
      }

      await loadFiles();
    } catch (err: any) {
      alert(err.message);
    } finally {
      setProcessingId("");
    }
  }

  const filteredFiles = useMemo(() => {
    return files.filter((file) => {
      const value = search.toLowerCase();

      return (
        file.fileName.toLowerCase().includes(value) ||
        file.sender.toLowerCase().includes(value) ||
        file.receiver.toLowerCase().includes(value)
      );
    });
  }, [files, search]);

  if (loading) {
    return (
      <main className="flex min-h-screen items-center justify-center bg-slate-950 text-white">

        <div className="flex items-center gap-4">

          <Loader2 className="h-7 w-7 animate-spin text-cyan-400" />

          <span className="text-lg">
            Loading intercepted files...
          </span>

        </div>

      </main>
    );
  }

  return (
    <main className="min-h-screen bg-slate-950 p-8 text-white">

      <div className="mx-auto max-w-7xl">

        <div className="mb-10 flex flex-col gap-5 md:flex-row md:items-center md:justify-between">

          <div>

            <h1 className="text-4xl font-bold">
              Attacker Dashboard
            </h1>

            <p className="mt-2 text-slate-400">
              Simulate a man-in-the-middle attack on files currently in transit.
            </p>

          </div>

          <button
            onClick={loadFiles}
            className="inline-flex items-center rounded-xl bg-cyan-500 px-5 py-3 font-semibold text-slate-950 hover:bg-cyan-400"
          >
            <RefreshCw className="mr-2 h-5 w-5" />

        </div>

        <div className="mb-8 flex items-center rounded-xl border border-slate-700 bg-slate-900 px-4 py-3">

          <Search className="mr-3 text-slate-500" />

          <input
            type="text"
            placeholder="Search file, sender or receiver..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className="w-full bg-transparent outline-none"
          />

        </div>

                <div className="overflow-hidden rounded-2xl border border-slate-800 bg-slate-900">

              <tr className="text-left">

                <th className="px-6 py-4">Receiver</th>

                <th className="px-6 py-4">Status</th>

                <th className="px-6 py-4">Encryption</th>

                <th className="px-6 py-4">Actions</th>

type PipelineStage =

function formatBytes(bytes: number) {
  if (!bytes) return "0 B";

  const units = ["B", "KB", "MB", "GB"];
  const index = Math.floor(
    Math.log(bytes) / Math.log(1024)
  );

  return `${(bytes / Math.pow(1024, index)).toFixed(
    index === 0 ? 0 : 1
  )} ${units[index] || "GB"}`;
}

function formatTime(value: string | null) {
  if (!value) return "—";

  return new Date(value).toLocaleTimeString([], {
    hour: "2-digit",
    minute: "2-digit",
    second: "2-digit",
  });
}

function getPipelineStage(
  transfer: Transfer
): PipelineStage {
  if (
    transfer.integrity === "COMPROMISED" ||
    transfer.attackAction === "MODIFIED"
  ) {
    return "compromised";
  }

  if (
    transfer.integrity === "VERIFIED" &&
    transfer.status === "DELIVERED"
  ) {
    return "receiver";
  }

  if (
    transfer.status === "VERIFYING" ||
    transfer.verifiedAt
  ) {
    return "integrity";
  }

  if (transfer.interceptedAt) {
    return "intercepted";
  }

  if (
    transfer.status === "IN_TRANSIT" ||
    transfer.status === "SENDING"
  ) {
    return "transmitting";
  }

  if (transfer.status === "ENCRYPTING") {
    return "encrypting";
  }

  return "sender";
}

function stageIndex(stage: PipelineStage) {
  const stages: PipelineStage[] = [
    "sender",
    "encrypting",
    "transmitting",
    "intercepted",
    "integrity",
    "receiver",
  ];

  if (stage === "compromised") return 5;

  return stages.indexOf(stage);
}

"use client";

import { useEffect, useState } from "react";

type User = {
  id: string;
  fullName: string;
  email: string;
  role: string;
  department: string | null;
  isBlocked: boolean;
  createdAt: string;
  _count: {
    sentFiles: number;
    receivedFiles: number;
    securityLogs: number;
  };
};

export default function AdminUserManagement() {
  const [users, setUsers] = useState<User[]>([]);
  const [loading, setLoading] = useState(true);
  const [processing, setProcessing] = useState<string | null>(null);

  async function loadUsers() {
    try {
      const response = await fetch("/api/admin/users", {
        cache: "no-store",
      });

      const data = await response.json();

      if (data.success) {
        setUsers(data.users);
      }
    } catch (error) {
      console.error("Failed to load users:", error);
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    loadUsers();

    const interval = setInterval(loadUsers, 3000);

    return () => clearInterval(interval);
  }, []);

  async function toggleBlock(user: User) {
    const action = user.isBlocked ? "unblock" : "block";

    setProcessing(user.id);

    try {
      const response = await fetch("/api/admin/users", {
        method: "PATCH",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          userId: user.id,
          action,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        alert(data.message || "Failed to update account.");
        return;
      }

      await loadUsers();
    } catch (error) {
      console.error(error);
      alert("Failed to update account.");
    } finally {
      setProcessing(null);
    }
  }

  async function deleteUser(user: User) {
    const confirmed = window.confirm(
      `Delete ${user.fullName}'s account?\n\nThis will permanently remove the account and its associated transfer records.`
    );

    if (!confirmed) {
      return;
    }

    setProcessing(user.id);

    try {
      const response = await fetch("/api/admin/users", {
        method: "DELETE",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          userId: user.id,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        alert(data.message || "Failed to delete account.");
        return;
      }

      await loadUsers();
    } catch (error) {
      console.error(error);
      alert("Failed to delete account.");
    } finally {
      setProcessing(null);
    }
  }

  return (
    <section className="mt-8 rounded-2xl border border-zinc-800 bg-zinc-950 p-6">
      <div className="mb-6 flex items-center justify-between">
        <div>
          <h2 className="text-xl font-semibold text-white">
            User Management
          </h2>
          <p className="mt-1 text-sm text-zinc-400">

          {users.length} Accounts
        </div>
      </div>

      ) : users.length === 0 ? (

        setError(err.message);

        setLoading(false);

  params: Promise<{
    id: string;

  params,
